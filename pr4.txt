3. Упрощенная версия generate_excel_receipt:
python
def generate_excel_receipt(cart, items, delivery_address, total_price):
    """
    Создает Excel-файл с чеком
    """
    wb = Workbook()
    ws = wb.active
    ws.title = f'Чек_{cart.id}'
    
    # Заголовок
    ws['A1'] = f'ЧЕК №{cart.id}'
    ws['A2'] = f'Покупатель: {cart.user.username}'
    ws['A3'] = f'Email: {cart.user.email}'
    ws['A4'] = f'Адрес: {delivery_address}'
    
    # Заголовки таблицы
    ws['A6'] = 'Товар'
    ws['B6'] = 'Цена'
    ws['C6'] = 'Кол-во'
    ws['D6'] = 'Сумма'
    
    # Заполняем товары
    row = 7
    for item in items:
        ws[f'A{row}'] = item.product.name
        ws[f'B{row}'] = float(item.product.price)
        ws[f'C{row}'] = item.quantity
        ws[f'D{row}'] = float(item.product.price * item.quantity)
        row += 1
    
    # Итого
    ws[f'C{row}'] = 'ИТОГО:'
    ws[f'D{row}'] = float(total_price)
    
    # Сохраняем в памяти
    excel_file = BytesIO()
    wb.save(excel_file)
    excel_file.seek(0)
    return excel_file
4. Упрощенная версия checkout:
python
@login_required
def checkout(request):
    # Получаем корзину и товары пользователя
    cart = Cart.objects.get(user=request.user)
    items = CartItem.objects.filter(cart=cart)
    
    # Если корзина пуста - возвращаем
    if not items:
        messages.warning(request, 'Корзина пуста')
        return redirect('cart_view')
    
    # Вычисляем общую сумму
    total_price = 0
    for item in items:
        item.total = item.product.price * item.quantity  # добавляем временное поле
        total_price += item.total
    
    # Обработка отправки формы
    if request.method == 'POST':
        # Получаем адрес из формы
        address = request.POST.get('delivery_address')
        
        if not address:
            messages.error(request, 'Укажите адрес доставки')
            return render(request, 'checkout.html', {
                'cart_items': items,
                'total_price': total_price
            })
        
        # Проверяем наличие на складе
        for item in items:
            if item.quantity > item.product.stock:
                messages.error(request, f'Недостаточно {item.product.name}')
                return render(request, 'checkout.html', {
                    'cart_items': items,
                    'total_price': total_price
                })
        
        # Создаем Excel-чек
        excel_file = generate_excel_receipt(cart, items, address, total_price)
        
        # Отправляем email
        email = EmailMessage(
            subject=f'Заказ №{cart.id}',
            body=f'Ваш заказ оформлен. Чек во вложении.',
            from_email='from@example.com',
            to=[request.user.email],
        )
        email.attach(f'receipt_{cart.id}.xlsx', excel_file.read(), 
                    'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet')
        email.send()
        
        # Обновляем остатки
        for item in items:
            product = item.product
            product.stock -= item.quantity
            product.save()
        
        # Очищаем корзину
        items.delete()
        
        messages.success(request, 'Заказ оформлен! Чек отправлен на почту.')
        return redirect('cart_view')
    
    # GET запрос - показываем форму
    return render(request, 'checkout.html', {
        'cart_items': items,
        'total_price': total_price
    })

6. Максимально упрощенная версия для понимания:
python
@login_required
def checkout(request):
    # 1. Берем корзину пользователя
    cart = Cart.objects.get(user=request.user)
    items = CartItem.objects.filter(cart=cart)
    
    # 2. Считаем общую сумму
    total = 0
    for item in items:
        item.sum = item.product.price * item.quantity  # сумма за одну позицию
        total += item.sum  # общая сумма
    
    # 3. Если нажали кнопку "Оформить"
    if request.method == 'POST':
        address = request.POST.get('delivery_address')
        
        # 4. Отправляем email (упрощенно)
        email = EmailMessage(
            'Заказ', 
            f'Адрес: {address}, Сумма: {total}', 
            'from@mail.com', 
            [request.user.email]
        )
        email.send()
        
        # 5. Чистим корзину
        items.delete()
        
        return redirect('cart_view')
    
    # 6. Показываем страницу с формой
    return render(request, 'checkout.html', {
        'items': items,
        'total': total
    })
Ключевые моменты:

items - это QuerySet с объектами CartItem

item.total (или item.sum) - это временное поле, которое мы сами создаем для удобства

После оформления заказа мы удаляем все items (очищаем корзину)

Excel-файл создается в памяти через BytesIO, не сохраняется на диск
